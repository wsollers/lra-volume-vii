\section{Languages and Structures}

\begin{toolkitbox}{L-Structures Toolkit --- Quick Reference}
\small
\begin{tabular}{l l}
\toprule
\textbf{Name}  & \textbf{Meaning} \\
\midrule
\hyperref[def:first-order-language]{First-order language} & Vocabulary or signature $\mathcal{L}$ \\
\hyperref[def:l-structure]{L-structure} & Interpretations of $\mathcal{L}$ on a universe $M$ \\
\hyperref[def:l-structure-reduct]{Reduct} & Forget symbols outside a sublanguage \\
\hyperref[def:l-structure-expansion]{Expansion} & Add interpretations for a larger language \\
\hyperref[def:l-structure-homomorphism]{Homomorphism} & Preserve functions, constants, and relations forward \\
\hyperref[def:l-structure-embedding]{Embedding} & Injective map preserving and reflecting relations \\
\hyperref[def:l-structure-isomorphism]{Isomorphism} & Bijective embedding \\
\hyperref[def:l-structure-automorphism]{Automorphism} & Isomorphism from a structure to itself \\
\bottomrule
\end{tabular}
\end{toolkitbox}
