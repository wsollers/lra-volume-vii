\section{Languages and Structures}

\begin{toolkitbox}{L-Structures Toolkit --- Quick Reference}
\small
\begin{tabular}{l l}
\toprule
\textbf{Concept}  & \textbf{Symbol / Notation} \\
\midrule
\hyperref[def:first-order-language]{First-order language}     & $\mathcal{L}$                          \\
\hyperref[def:l-structure]{L-structure}              & $\mathcal{M} = (M, \mathcal{L}^{\mathcal{M}})$ \\
\hyperref[def:l-structure]{Domain / universe}        & $|\mathcal{M}|$ or $M$                 \\
\hyperref[def:l-structure]{Interpretation of $f$}    & $f^{\mathcal{M}} : M^n \to M$          \\
\hyperref[def:l-structure]{Interpretation of $R$}    & $R^{\mathcal{M}} \subseteq M^n$        \\
\hyperref[def:l-structure]{Interpretation of $c$}    & $c^{\mathcal{M}} \in M$                \\
\hyperref[def:variable-assignment]{Variable assignment}      & $s : \mathrm{Var} \to M$               \\
\hyperref[def:term-evaluation]{Term evaluation}          & $s(t) \in M$                           \\
\hyperref[def:satisfaction]{Satisfaction}             & $\mathcal{M} \models \varphi[s]$       \\
\bottomrule
\end{tabular}
\end{toolkitbox}
