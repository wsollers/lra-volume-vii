\begin{proposition*}[Isomorphisms Preserve Formulas]
Let $i\colon\mathcal{M}\to\mathcal{N}$ be an isomorphism of
$\mathcal{L}$-structures. For every $\mathcal{L}$-formula $\varphi(\bar x)$
and assignment $s\colon\mathrm{Var}\to M$,
\[
  \mathcal{M}\models \varphi[s]
  \quad\text{iff}\quad
  \mathcal{N}\models \varphi[i\circ s].
\]
\end{proposition*}
